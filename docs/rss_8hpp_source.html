97template<std::size_t accumulators_ = 4, typename Input_, typename Output_>
98RssResult<Output_> rss(const std::size_t num_total, const std::size_t num_non_zero, const Input_* const ptr, RssWorkspace<Output_>& work, const RssOptions<Output_>& options) {
99 static_assert(std::is_floating_point<Output_>::value);
100
101 RssResult<Output_> output;
102 if (num_total == 0) {
103 output.mean = options.mean_placeholder;
104 return output;
105 }
106
107 Output_& mean = output.mean;
108 PairwiseSumOptions psopt;
109 psopt.max_sum_length = options.max_sum_length;
110 mean = pairwise_sum<accumulators_>(num_non_zero, ptr, work.pswork, psopt);
111 mean /= num_total;
112
113 Output_& ssd = output.rss;
114 ssd = pairwise_sum_abstract<accumulators_>(
115 num_non_zero,
116 [&](const std::size_t i) -> Output_ {
117 const auto delta = static_cast<Output_>(ptr[i]) - mean;
118 return delta * delta;
119 },
120 work.pswork,
121 psopt
122 );
123
124 assert(num_non_zero <= num_total);
125 if (num_non_zero < num_total) {
126 ssd += static_cast<Output_>(num_total - num_non_zero) * mean * mean;
127 }
128
129 return output;
130}

195template<typename Output_ = double, typename Value_, typename Count_>
196void welford_add(Output_& mean, Output_& sumsq, const Value_ value, const Count_ count) {
197 Output_ delta = static_cast<Output_>(value) - mean;
198 mean += delta / count;
199 sumsq += delta * (static_cast<Output_>(value) - mean);

401void welford_add_zeros(Output_& mean, Output_& sumsq, const Count_ num_total, const Count_ num_non_zero) {
402 assert(num_total > 0);
403 const auto ratio = static_cast<Output_>(num_non_zero) / static_cast<Output_>(num_total);
404 sumsq += mean * mean * ratio * (num_total - num_non_zero);

577 template<typename Index_, class Skip_>
578 void add(const std::size_t num_non_zero_obs, const Input_* value, const Index_* index, Skip_ skip) {
579 static_assert(std::is_integral<Index_>::value);
580
581 // my_count is the upper bound of all my_num_non_zero, so no need to check individual increments.
582 my_count = sanisizer::sum<Count_>(my_count, 1);
583
584 for (std::size_t i = 0; i < num_non_zero_obs; ++i) {
585 const auto val = value[i];
586 const auto ri = index[i];
587 if (skip(ri, val)) {
588 ++my_num_unskipped[ri]; // storing the number that was skipped so we don't have to add the zeros later.
589 } else {
590 welford_add(my_mean[ri], my_rss[ri], val, ++(my_num_non_zero[ri]));
591 }

611 for (std::size_t i = 0; i < my_num_obj; ++i) {
612 my_num_unskipped[i] = my_count - my_num_unskipped[i];
613 }
614
615 if (my_count == 0) {

708template<typename Float_>
709Float_ rss_to_variance(const std::size_t num_total, const Float_ rss) {
710 if (num_total < 1) {
711 return std::numeric_limits<Float_>::quiet_NaN();

728template<typename Float_>
729void rss_to_variance(const std::size_t num_obj, const std::size_t num_total, Float_* const rss) {
730 if (num_total < 1) {
731 std::fill_n(rss, num_obj, std::numeric_limits<Float_>::quiet_NaN());
732 } else {
733 // For consistency with the other overloads, we won't do the '* (1/denom)' trick.
734 // It shouldn't have much effect on throughput anyway as the bottleneck should be reading from memory.
735 for (std::size_t i = 0; i < num_obj; ++i) {
736 rss[i] /= num_total - 1;
737 }
738 }
739}
