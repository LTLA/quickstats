98template<std::size_t accumulators_ = 4, typename Input_, typename Output_>
99RssResult<Output_> rss(const std::size_t num_total, const std::size_t num_non_zero, const Input_* const ptr, RssWorkspace<Output_>& work, const RssOptions<Output_>& options) {
100 static_assert(std::is_floating_point<Output_>::value);
101
102 RssResult<Output_> output;
103 if (num_total == 0) {
104 output.mean = options.mean_placeholder;
105 return output;
106 }
107
108 Output_& mean = output.mean;
109 PairwiseSumOptions psopt;
110 psopt.max_sum_length = options.max_sum_length;
111 mean = pairwise_sum<accumulators_>(num_non_zero, ptr, work.pswork, psopt);
112 mean /= num_total;
113
114 Output_& ssd = output.rss;
115 ssd = pairwise_sum_abstract<accumulators_>(
116 num_non_zero,
117 [&](const std::size_t i) -> Output_ {
118 const auto delta = static_cast<Output_>(ptr[i]) - mean;
119 return delta * delta;
120 },
121 work.pswork,
122 psopt
123 );
124
125 assert(num_non_zero <= num_total);
126 if (num_non_zero < num_total) {
127 ssd += static_cast<Output_>(num_total - num_non_zero) * mean * mean;
128 }
129
130 return output;
131}

161void welford_add(Output_& mean, Output_& sumsq, const Value_ value, const Count_ count) {
162 Output_ delta = static_cast<Output_>(value) - mean;
163 mean += delta / count;
164 sumsq += delta * (static_cast<Output_>(value) - mean);

366void welford_add_zeros(Output_& mean, Output_& sumsq, const Count_ num_total, const Count_ num_non_zero) {
367 assert(num_total > 0);
368 const auto ratio = static_cast<Output_>(num_non_zero) / static_cast<Output_>(num_total);
369 sumsq += mean * mean * ratio * (num_total - num_non_zero);
370 mean *= ratio;

544 static_assert(std::is_integral<Index_>::value);
545

693template<typename Float_>
694void rss_to_variance(const std::size_t num_obj, const std::size_t num_total, Float_* const rss) {
695 if (num_total < 1) {
696 std::fill_n(rss, num_obj, std::numeric_limits<Float_>::quiet_NaN());
697 } else {
698 // For consistency with the other overloads, we won't do the '* (1/denom)' trick.
699 // It shouldn't have much effect on throughput anyway as the bottleneck should be reading from memory.
700 for (std::size_t i = 0; i < num_obj; ++i) {
701 rss[i] /= num_total - 1;
702 }
703 }
704}
