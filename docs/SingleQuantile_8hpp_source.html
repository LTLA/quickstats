27template<typename Output_>
28void configure_single_quantile(
29 const Output_ quantile,
30 const std::size_t num_m1,
31 std::size_t& upper_index,
32 Output_& upper_fraction,
33 bool& skip_lower
34) {
35 if (quantile < 0 || quantile > 1) {
36 throw std::out_of_range("'quantile' should lie in [0, 1]");
37 }
38
39 const Output_ raw_index = static_cast<Output_>(num_m1) * quantile;
40 const Output_ raw_upper_index = std::ceil(raw_index);
41 const Output_ raw_lower_index = std::floor(raw_index);
42
43 // Protect the cast from conversion imprecision between size_t and Float_,
44 // e.g., if it rounds up or if it converts it to an Inf.
45 const auto converted_upper_index = sanisizer::from_float<std::size_t>(raw_upper_index);
46 if (converted_upper_index <= num_m1) {
47 upper_index = converted_upper_index;
48 upper_fraction = raw_index - raw_lower_index;
49 skip_lower = (raw_upper_index == raw_lower_index);
50 } else {
51 // Just gently cap it at the maximum value.
52 upper_index = num_m1;
53 upper_fraction = 0;
54 skip_lower = true;
55 }
